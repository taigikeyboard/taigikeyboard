### `engine/lexicon/src/classification.rs` — sketch

```rust
//! Input classification primitives for the IME and Tab3 search.
//!
//! Pure functions. No I/O, no engine handle. Two consumers:
//! - `classify_input` — IME autocomplete classifier (was iOS
//!   `AutocompleteInputClassifier.swift` / Android `.kt`).
//! - `is_hanzi` — Tab3 search short-circuit (was iOS
//!   `CandidateProcessor.isHanzi` / Android Tab3 inline check).
//!
//! See `INVARIANT_LEX_INPUT_CLASSIFICATION` and subcases in
//! `docs/architecture/behavioral-invariants.md`.

use phonetics::{contains_tps, has_tone_marks, tps_to_tl};
use protos::engine::{ClassifyInputResponse, InputType};

/// True iff `text` contains any CJK Unified Ideograph (Unified +
/// Extensions A-E). Extensions F/G/H/I/J are intentionally excluded —
/// see `INVARIANT_LEX_INPUT_CLASSIFICATION_HANZI_RANGE`.
pub fn is_hanzi(text: &str) -> bool {
    text.chars().any(|c| {
        let cp = c as u32;
        (0x4E00..=0x9FFF).contains(&cp)        // CJK Unified
            || (0x3400..=0x4DBF).contains(&cp)  // Extension A
            || (0x20000..=0x2A6DF).contains(&cp) // Extension B
            || (0x2A700..=0x2B73F).contains(&cp) // Extension C
            || (0x2B740..=0x2B81F).contains(&cp) // Extension D
            || (0x2B820..=0x2CEAF).contains(&cp) // Extension E
    })
}

/// True iff `text` contains an ASCII numeric tone digit.
///
/// ASCII digits 2,3,5,6,7,8,9 are numeric tone markers; 1, 4, and 0
/// are not. See `INVARIANT_LEX_INPUT_CLASSIFICATION_NUMERIC_TONE_SET`.
pub fn contains_numeric_tone(text: &str) -> bool {
    text.chars().any(|c| c.is_ascii_digit() && !matches!(c, '0' | '1' | '4'))
}

/// Classify `raw` into `(InputType, search_key)`.
///
/// Precedence — see
/// `INVARIANT_LEX_INPUT_CLASSIFICATION_PRECEDENCE`.
/// Search key — see
/// `INVARIANT_LEX_INPUT_CLASSIFICATION_SEARCH_KEY`.
pub fn classify_input(raw: &str) -> ClassifyInputResponse {
    let input_type = if is_hanzi(raw) {
        InputType::Hanzi
    } else if has_tone_marks(raw) {
        InputType::RomanWithTone
    } else if contains_numeric_tone(raw) {
        InputType::RomanWithTone
    } else {
        InputType::RomanNoTone
    };

    let search_key = if contains_tps(raw) {
        tps_to_tl(raw)
    } else {
        raw.to_owned()
    };

    ClassifyInputResponse {
        input_type: input_type as i32,
        search_key,
    }
}
```

## 3. Bridge changes

### 3.1 iOS — `RustEngineBridge+Lexicon.swift`

Add to existing extension (per Codex round 2 D6 — don't bloat base bridge):

```swift
public extension RustEngineBridge {
    struct ClassificationResult {
        public let inputType: InputType  // existing platform enum
        public let searchKey: String
    }

    /// Classify rawInput → (InputType, searchKey). Engine resolves precedence
    /// + builds search_key (TPS→TL conversion when applicable).
    static func classifyInput(_ raw: String) -> ClassificationResult {
        var req = Taigi_Engine_ClassifyInputRequest()
        req.raw = raw
        let resp = lexiconDispatch(method: .classifyInput(req), op: "classifyInput")
        guard case let .classifyInputResult(result)? = resp?.result else {
            recordFailure(op: "classifyInput", message: "missing result")
            return ClassificationResult(inputType: .romanWithoutTone, searchKey: raw)
        }
        return ClassificationResult(
            inputType: platformInputType(from: result.inputType),
            searchKey: result.searchKey,
        )
    }

    /// Low-level: true iff `text` contains a CJK codepoint (A-E ranges).
    /// Used by Tab3 search short-circuit.
    static func isHanzi(_ text: String) -> Bool {
        var req = Taigi_Engine_IsHanziRequest()
        req.text = text
        let resp = lexiconDispatch(method: .isHanzi(req), op: "isHanzi")
        guard case let .isHanziResult(result)? = resp?.result else { return false }
        return result.isHanzi
    }

    /// Local mapping `Taigi_Engine_InputType` → platform `InputType`.
    /// (No `fromProto` exists today; add this helper alongside the bridge.)
    private static func platformInputType(from proto: Taigi_Engine_InputType) -> InputType {
        switch proto {
        case .hanzi:           return .hanzi
        case .romanWithTone:   return .romanWithTone
        case .romanNoTone:     return .romanWithoutTone
        case .unspecified, .UNRECOGNIZED: return .romanWithoutTone
        }
    }
}
```

### 3.2 Android — `engine/LexiconBridge.kt`

Methods land in the existing top-level `LexiconBridge` object (file
`android/app/src/main/java/com/siansiansu/taigikeyboard/engine/LexiconBridge.kt`).
Return types use the platform `ime.dictionary.InputType` enum directly so
callers don't need to round-trip through a bridge-only enum.

```kotlin
import com.siansiansu.taigikeyboard.engine.proto.ClassifyInputRequest
import com.siansiansu.taigikeyboard.engine.proto.IsHanziRequest
import com.siansiansu.taigikeyboard.engine.proto.LexiconRequest
import com.siansiansu.taigikeyboard.ime.dictionary.InputType

data class ClassificationResult(
    val inputType: InputType,
    val searchKey: String,
)

fun classifyInput(raw: String): ClassificationResult {
    val payload = ClassifyInputRequest.newBuilder().setRaw(raw).build()
    val resp = dispatch(LexiconRequest.newBuilder().setClassifyInput(payload).build())
        ?: return ClassificationResult(InputType.RomanWithoutTone, raw)
    if (!resp.hasClassifyInputResult()) {
        return ClassificationResult(InputType.RomanWithoutTone, raw)
    }
    val result = resp.classifyInputResult
    return ClassificationResult(
        inputType = platformInputType(result.inputType),
        searchKey = result.searchKey,
    )
}

fun isHanzi(text: String): Boolean {
    val payload = IsHanziRequest.newBuilder().setText(text).build()
    val resp = dispatch(LexiconRequest.newBuilder().setIsHanzi(payload).build()) ?: return false
    if (!resp.hasIsHanziResult()) return false
    return resp.isHanziResult.isHanzi
}

// Private mapper proto::InputType → ime.dictionary.InputType.
// (No fromProto exists today; add alongside the bridge methods.)
private fun platformInputType(
    proto: com.siansiansu.taigikeyboard.engine.proto.InputType,
): InputType = when (proto) {
    com.siansiansu.taigikeyboard.engine.proto.InputType.INPUT_TYPE_HANZI           -> InputType.Hanzi
    com.siansiansu.taigikeyboard.engine.proto.InputType.INPUT_TYPE_ROMAN_WITH_TONE -> InputType.RomanWithTone
    com.siansiansu.taigikeyboard.engine.proto.InputType.INPUT_TYPE_ROMAN_NO_TONE,
    com.siansiansu.taigikeyboard.engine.proto.InputType.INPUT_TYPE_UNSPECIFIED,
    com.siansiansu.taigikeyboard.engine.proto.InputType.UNRECOGNIZED               -> InputType.RomanWithoutTone
}
```
